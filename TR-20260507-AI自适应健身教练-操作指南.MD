## 2. 痛点研究与客户需求

### 2.1 核心痛点汇总 (来自Reddit/Twitter/HN/媒体深度调研)

#### 痛点 #1: 静态计划无法适应每日状态

**严重程度**: 🔴 极高

> "Every fitness app I've tried gives me a static program. If I'm having a strong day, it doesn't push me harder. If I'm exhausted, it doesn't scale back."
> — iwishtherewas.xyz用户

> "I kept thinking: why can't I just tell my workout app how I'm feeling and have it figure out the rest?"
> — Adapt Fit AI创始人 (Medium)

**用户需求**: 应用必须能根据每日能量水平/恢复状态实时调整训练强度和容量

#### 痛点 #2: AI教练"失忆"问题

**严重程度**: 🔴 极高

> "Gemini kept forgetting things. It forgot my training plan altogether... The only way to find my complete training plan was to scroll back through months of conversations."
> — Android Authority

> "It even forgot that I'm a vegetarian and listed a meal plan full of chicken and other meats."
> — Android Authority

**用户需求**: 所有训练数据必须持久化存储, AI决策基于结构化数据而非对话上下文

#### 痛点 #3: AI教练给出错误/危险建议

**严重程度**: 🔴 极高

> "Runna Pushed Some Runners Into Injuries. The Virtual-Coach App Is Now Easing Up."
> — The Verge

> "Even that simple attempt to add a warmup was a comedy of errors, especially when I asked it to keep the workouts to 30 minutes. It can't do math."
> — Lifehacker (Fitbit AI评测)

**用户需求**: 算法必须基于运动科学原理(渐进超负荷/双重进阶), 而非LLM生成式输出; 必须有安全边界

#### 痛点 #4: 缺乏长期规划能力

**严重程度**: 🟠 高

> "There was no way to view next week, or the week after that, or to see how many weeks were even in this alleged plan."
> — PCMag (Google AI Coach评测)

**用户需求**: 必须提供周期化训练计划(含deload周), 可查看未来数周安排

#### 痛点 #5: 伤害/疼痛无法适配

**严重程度**: 🟠 高

> "Fitness apps don't care if you're injured. Mine does."
> — Adaptifit创始人 (Reddit)

> "Yo, my knee hurts on squats — what do I swap in?"
> — Reddit用户期望

**用户需求**: 标记疼痛部位后, 自动替换相关动作为替代动作

#### 痛点 #6: 过度复杂/功能臃肿

**严重程度**: 🟡 中

> "All of those apps seem to have become so bloated... It almost seems to have made planning more complicated."
> — Reddit r/travel (类比旅行应用, 同样适用于健身应用)

**用户需求**: 极简界面, 核心功能一目了然, 不需要47个问题的问卷

#### 痛点 #7: 需要过多"手把手"指导AI

**严重程度**: 🟡 中

> "I was spending so much time coaching various AI tools on how to coach me that I'd begun dreading my workouts."
> — The Verge

**用户需求**: 零配置启动, 开箱即用, AI自动学习用户模式

### 4.2 ReadinessScore Model — 每日准备度评分

```swift
import Foundation
import SwiftData

@Model
final class ReadinessScore {
    @Attribute(.unique) var id: UUID
    var date: Date
    var energyLevel: EnergyLevel
    var sleepQuality: SleepQuality
    var sorenessLevel: SorenessLevel
    var motivationLevel: MotivationLevel
    var overallScore: Double
    var healthKitSleepHours: Double?
    var healthKitHRV: Double?
    var healthKitRestingHR: Double?

    enum EnergyLevel: Int, Codable, CaseIterable {
        case exhausted = 1
        case low = 2
        case moderate = 3
        case high = 4
        case peaked = 5

        var label: String {
            switch self {
            case .exhausted: "Exhausted"
            case .low: "Low Energy"
            case .moderate: "Okay"
            case .high: "Energized"
            case .peaked: "Feeling Great"
            }
        }
    }

    enum SleepQuality: Int, Codable, CaseIterable {
        case terrible = 1
        case poor = 2
        case fair = 3
        case good = 4
        case excellent = 5
    }

    enum SorenessLevel: Int, Codable, CaseIterable {
        case none = 1
        case mild = 2
        case moderate = 3
        case significant = 4
        case severe = 5
    }

    enum MotivationLevel: Int, Codable, CaseIterable {
        case zero = 1
        case low = 2
        case moderate = 3
        case high = 4
        case fired = 5
    }

    init(
        energyLevel: EnergyLevel,
        sleepQuality: SleepQuality,
        sorenessLevel: SorenessLevel,
        motivationLevel: MotivationLevel,
        healthKitSleepHours: Double? = nil,
        healthKitHRV: Double? = nil,
        healthKitRestingHR: Double? = nil
    ) {
        self.id = UUID()
        self.date = Date()
        self.energyLevel = energyLevel
        self.sleepQuality = sleepQuality
        self.sorenessLevel = sorenessLevel
        self.motivationLevel = motivationLevel
        self.healthKitSleepHours = healthKitSleepHours
        self.healthKitHRV = healthKitHRV
        self.healthKitRestingHR = healthKitRestingHR
        self.overallScore = Self.calculateScore(
            energy: energyLevel,
            sleep: sleepQuality,
            soreness: sorenessLevel,
            motivation: motivationLevel
        )
    }

    static func calculateScore(
        energy: EnergyLevel,
        sleep: SleepQuality,
        soreness: SorenessLevel,
        motivation: MotivationLevel
    ) -> Double {
        let energyWeight = 0.30
        let sleepWeight = 0.30
        let sorenessWeight = 0.25
        let motivationWeight = 0.15

        let rawScore = (
            Double(energy.rawValue) * energyWeight +
            Double(sleep.rawValue) * sleepWeight +
            (6.0 - Double(soreness.rawValue)) * sorenessWeight +
            Double(motivation.rawValue) * motivationWeight
        )

        return min(max(rawScore / 5.0 * 100.0, 0), 100)
    }
}
```

### 4.3 ReadinessEngine — 核心自适应引擎

```swift
import Foundation

struct ReadinessEngine {
    enum TrainingAdjustment {
        case scaleUp
        case maintain
        case scaleDown
        case deload
        case rest

        var volumeMultiplier: Double {
            switch self {
            case .scaleUp: 1.15
            case .maintain: 1.0
            case .scaleDown: 0.75
            case .deload: 0.5
            case .rest: 0.0
            }
        }

        var intensityMultiplier: Double {
            switch self {
            case .scaleUp: 1.05
            case .maintain: 1.0
            case .scaleDown: 0.85
            case .deload: 0.7
            case .rest: 0.0
            }
        }

        var rpeAdjustment: Int {
            switch self {
            case .scaleUp: 1
            case .maintain: 0
            case .scaleDown: -2
            case .deload: -3
            case .rest: 0
            }
        }

        var description: String {
            switch self {
            case .scaleUp: "You're recovered and ready to push! Increasing volume and intensity."
            case .maintain: "On track. Maintaining planned progression."
            case .scaleDown: "Your body needs a lighter session. Reducing volume and intensity."
            case .deload: "Signs of fatigue detected. Taking a deload to recover."
            case .rest: "Your body needs rest today. Recovery is part of the process."
            }
        }
    }

    static func evaluate(readiness: ReadinessScore) -> TrainingAdjustment {
        let score = readiness.overallScore

        if score >= 80 { return .scaleUp }
        if score >= 60 { return .maintain }
        if score >= 40 { return .scaleDown }
        if score >= 25 { return .deload }
        return .rest
    }

    static func evaluateWithHealthKit(
        readiness: ReadinessScore,
        recentPerformance: [SetRecord]
    ) -> TrainingAdjustment {
        let baseAdjustment = evaluate(readiness: readiness)

        var fatigueSignals = 0
        if let hrv = readiness.healthKitHRV, let restingHR = readiness.healthKitRestingHR {
            if restingHR > 70 { fatigueSignals += 1 }
            if hrv < 40 { fatigueSignals += 1 }
        }

        let recentFailedSets = recentPerformance.filter { $0.completedReps < $0.targetReps }
        if Double(recentFailedSets.count) / Double(max(recentPerformance.count, 1)) > 0.4 {
            fatigueSignals += 1
        }

        if fatigueSignals >= 2 {
            switch baseAdjustment {
            case .scaleUp: return .maintain
            case .maintain: return .scaleDown
            case .scaleDown: return .deload
            default: return baseAdjustment
            }
        }

        return baseAdjustment
    }
}
```

### 4.4 AdaptiveProgressionService — 自适应渐进超负荷

```swift
import Foundation
import SwiftData

struct AdaptiveProgressionService {
    struct ProgressionSuggestion {
        let targetWeight: Double
        let targetReps: Int
        let targetSets: Int
        let targetRPE: Int
        let reasoning: String
    }

    static func suggestNextWorkout(
        for exercise: Exercise,
        lastSessionRecords: [SetRecord],
        adjustment: ReadinessEngine.TrainingAdjustment,
        plan: TrainingPlan
    ) -> ProgressionSuggestion {
        guard let bestSet = lastSessionRecords.max(by: { $0.weight < $1.weight }) else {
            return ProgressionSuggestion(
                targetWeight: exercise.startingWeight ?? 45.0,
                targetReps: 8,
                targetSets: 3,
                targetRPE: 7,
                reasoning: "Starting with baseline weight for new exercise."
            )
        }

        let baseWeight = bestSet.weight
        let baseReps = bestSet.completedReps
        let repRange = plan.repRange

        let weightIncrement = calculateWeightIncrement(for: baseWeight)
        let volumeMultiplier = adjustment.volumeMultiplier
        let intensityMultiplier = adjustment.intensityMultiplier
        let rpeAdjustment = adjustment.rpeAdjustment

        var targetReps: Int
        var targetWeight: Double
        var targetSets: Int
        var targetRPE: Int
        var reasoning: String

        if baseReps >= repRange.upperBound {
            targetWeight = baseWeight + weightIncrement
            targetReps = repRange.lowerBound
            reasoning = "You hit \(baseReps) reps at \(baseWeight)lbs — increasing weight by \(weightIncrement)lbs."
        } else if baseReps >= repRange.lowerBound {
            targetWeight = baseWeight
            targetReps = baseReps + 1
            reasoning = "Adding 1 rep at \(baseWeight)lbs to drive progression within the range."
        } else {
            targetWeight = max(baseWeight - weightIncrement, 0)
            targetReps = repRange.lowerBound
            reasoning = "Below target reps — reducing weight by \(weightIncrement)lbs to rebuild."
        }

        targetWeight = round(targetWeight * intensityMultiplier / weightIncrement) * weightIncrement
        targetReps = max(Int(Double(targetReps) * volumeMultiplier), 1)
        targetSets = max(Int(Double(plan.setsPerExercise) * volumeMultiplier), 1)
        targetRPE = min(max(plan.baseRPE + rpeAdjustment, 5), 10)

        reasoning += " " + adjustment.description

        return ProgressionSuggestion(
            targetWeight: targetWeight,
            targetReps: targetReps,
            targetSets: targetSets,
            targetRPE: targetRPE,
            reasoning: reasoning
        )
    }

    static func calculateWeightIncrement(for currentWeight: Double) -> Double {
        if currentWeight < 50 { return 2.5 }
        if currentWeight < 100 { return 5.0 }
        if currentWeight < 200 { return 10.0 }
        return 10.0
    }
}
```

### 4.5 PlanGenerator — 周期化训练计划生成

```swift
import Foundation

struct PlanGenerator {
    struct PlanConfig {
        let goal: FitnessGoal
        let experienceLevel: ExperienceLevel
        let daysPerWeek: Int
        let sessionDuration: Int
        let availableEquipment: Set<Equipment>
        let muscleGroups: [MuscleGroup]
    }

    enum FitnessGoal: String, CaseIterable {
        case strength = "Strength"
        case hypertrophy = "Muscle Growth"
        case endurance = "Endurance"
        case general = "General Fitness"
    }

    enum ExperienceLevel: String, CaseIterable {
        case beginner = "Beginner"
        case intermediate = "Intermediate"
        case advanced = "Advanced"
    }

    enum Equipment: String, CaseIterable {
        case barbell, dumbbell, machine, cable, kettlebell, bodyweight, resistanceBand
    }

    enum MuscleGroup: String, CaseIterable {
        case chest, back, shoulders, biceps, triceps, quads, hamstrings, glutes, core, calves
    }

    static func generatePlan(config: PlanConfig) -> TrainingPlan {
        let split = determineSplit(daysPerWeek: config.daysPerWeek)
        let phases = generatePhases(
            goal: config.goal,
            experience: config.experienceLevel,
            split: split,
            equipment: config.availableEquipment
        )

        return TrainingPlan(
            name: "\(config.goal.rawValue) Plan",
            split: split,
            phases: phases,
            totalWeeks: calculateTotalWeeks(experience: config.experienceLevel),
            config: config
        )
    }

    static func determineSplit(daysPerWeek: Int) -> TrainingSplit {
        switch daysPerWeek {
        case 2: return .fullBody2x
        case 3: return .fullBody3x
        case 4: return .upperLower
        case 5: return .pushPullLegsUpperLower
        case 6: return .pushPullLegs2x
        default: return .fullBody3x
        }
    }

    static func generatePhases(
        goal: FitnessGoal,
        experience: ExperienceLevel,
        split: TrainingSplit,
        equipment: Set<Equipment>
    ) -> [TrainingPhase] {
        let baseWeeks = experience == .beginner ? 4 : (experience == .intermediate ? 3 : 2)

        switch goal {
        case .strength:
            return [
                TrainingPhase(name: "Foundation", weeks: baseWeeks, repRange: 8...12, baseRPE: 7, intensityMod: 0.85),
                TrainingPhase(name: "Build", weeks: baseWeeks, repRange: 5...8, baseRPE: 8, intensityMod: 0.90),
                TrainingPhase(name: "Peak", weeks: max(baseWeeks - 1, 2), repRange: 3...5, baseRPE: 9, intensityMod: 0.95),
                TrainingPhase(name: "Deload", weeks: 1, repRange: 8...12, baseRPE: 6, intensityMod: 0.60)
            ]
        case .hypertrophy:
            return [
                TrainingPhase(name: "Volume Block", weeks: baseWeeks, repRange: 8...12, baseRPE: 7, intensityMod: 0.80),
                TrainingPhase(name: "Intensity Block", weeks: baseWeeks, repRange: 6...10, baseRPE: 8, intensityMod: 0.87),
                TrainingPhase(name: "Deload", weeks: 1, repRange: 10...15, baseRPE: 6, intensityMod: 0.60)
            ]
        default:
            return [
                TrainingPhase(name: "General Training", weeks: baseWeeks, repRange: 8...12, baseRPE: 7, intensityMod: 0.82),
                TrainingPhase(name: "Deload", weeks: 1, repRange: 10...15, baseRPE: 6, intensityMod: 0.60)
            ]
        }
    }

    static func calculateTotalWeeks(experience: ExperienceLevel) -> Int {
        switch experience {
        case .beginner: return 12
        case .intermediate: return 10
        case .advanced: return 8
        }
    }
}
```

### 4.6 InjuryAdapter — 疼痛/伤害动作替换

```swift
import Foundation

struct InjuryAdapter {
    struct PainEntry {
        let bodyPart: BodyPart
        let severity: PainSeverity
        let movementsToAvoid: [String]
    }

    enum BodyPart: String, CaseIterable {
        case knee, lowerBack, shoulder, elbow, wrist, hip, ankle, neck
    }

    enum PainSeverity: String, CaseIterable {
        case mild, moderate, severe
    }

    private static let substitutionMap: [BodyPart: [String: String]] = [
        .knee: [
            "Barbell Back Squat": "Box Squat",
            "Leg Press": "Bulgarian Split Squat (Pain-Free Depth)",
            "Leg Extension": "Hip Thrust",
            "Running": "Cycling or Swimming"
        ],
        .lowerBack: [
            "Barbell Deadlift": "Rack Pull",
            "Barbell Row": "Chest-Supported Row",
            "Barbell Back Squat": "Goblet Squat",
            "Good Morning": "Glute Ham Raise"
        ],
        .shoulder: [
            "Overhead Press": "Landmine Press",
            "Bench Press": "Floor Press",
            "Pull-Up": "Lat Pulldown",
            "Lateral Raise": "Cable Lateral Raise (Lower Angle)"
        ],
        .elbow: [
            "Barbell Curl": "Hammer Curl",
            "Skull Crusher": "Pushdown",
            "Chin-Up": "Neutral Grip Pulldown"
        ],
        .wrist: [
            "Barbell Bench Press": "Dumbbell Bench Press",
            "Barbell Curl": "EZ-Bar Curl",
            "Front Squat": "Goblet Squat"
        ]
    ]

    static func substitute(
        exerciseName: String,
        painEntries: [PainEntry]
    ) -> (substitute: String, reason: String)? {
        for entry in painEntries {
            if let subs = substitutionMap[entry.bodyPart],
               let sub = subs[exerciseName] {
                let reason = "Swapped \(exerciseName) → \(sub) due to \(entry.bodyPart.rawValue) \(entry.severity.rawValue) pain."
                return (sub, reason)
            }
        }
        return nil
    }

    static func filterExercises(
        exercises: [Exercise],
        painEntries: [PainEntry]
    ) -> [Exercise] {
        var filtered = exercises
        for entry in painEntries {
            if entry.severity == .severe {
                let affectedMovements = substitutionMap[entry.bodyPart]?.keys ?? []
                filtered = filtered.filter { !affectedMovements.contains($0.name) }
            }
        }
        return filtered
    }
}
```
